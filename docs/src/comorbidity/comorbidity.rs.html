19use std::path::PathBuf;
20use clap::Parser;
21use table::NetStateArgs;
22use std::time::Instant;
23use color_eyre::{Result, eyre::Context};
24use random::rep3::Rep3State;
25use random::MpcState;
26use communication::rep3::id::PartyID;
27use protocols::rep3_ring::arithmetic::open;
28use net::tcp::{TcpNetwork, NetworkConfig};
29use experiments::secrecy_database_gen::{gen_cohort_table, gen_diagnosis_comorbidity_table};
30use experiments::net_statistics::install_tracing;
31use experiments::net_statistics::print_communication_stats;
32use table::table_operator::{Join, Groupby, AggFunc, OrderBy, Open, Project};
33use table::column_operator::PrefixSum;
34use polars::prelude::*;
35
36
37#[derive(Parser)]
38struct Args {
39    /// The config file path
40    #[clap(short = 'c', long, value_name = "CONFIGDIR")]
41    config_dir: PathBuf,
42    
43    /// The party ID (0, 1, 2)
44    #[clap(short = 'p', long, value_name = "PARTYID", default_value = "0")]
45    party_id: String,
46
47    /// The number of threads
48    #[clap(short = 't', long, value_name = "THREADS", default_value = "6")]
49    threads: usize,
50
51    /// Scale factor
52    #[clap(short = 's', long, value_name = "SF", default_value = "0.01")]
53    sf: f32,
54}
55
56
57fn main() -> Result<()> {
58    let args = Args::parse();
59    install_tracing();
60
61    let sf = args.sf;
62    let partyid = args.party_id.clone();
63    let default_threads = rayon::current_num_threads() / 2;
64
65    tracing::info!("setting up network");
66    let mut nets: Vec<TcpNetwork> = Vec::new();
67    let mut states: Vec<Rep3State> = Vec::new();
68
69    for i in 0..args.threads {
70        let file_path = PathBuf::from(format!("{}{}/config_party{}.toml", args.config_dir.display(), i, partyid));
71        let config: NetworkConfig = toml::from_str(&std::fs::read_to_string(file_path).context("opening config file")?).context("parsing config file")?;
72        let net = TcpNetwork::new(config)?;
73        let state = Rep3State::new(&net)?;
74        nets.push(net);
75        states.push(state);
76    }
77
78    if states.len() <= default_threads {
79        let diff = default_threads - states.len();
80        for _i in 0..diff {
81            let state = states[0].fork(0)?;
82            states.push(state);
83        }
84    }
85
86    let nets = nets.iter().collect::<Vec<&TcpNetwork>>();
87    let mut states = states.iter_mut().collect::<Vec<&mut Rep3State>>();
88    let party_id = states[0].id;
89
90    let mut mpc_exec_args = NetStateArgs::new(&nets, &mut states);
91    
92    tracing::info!("Network setup completed");
93
94
95    tracing::info!("Generating tables");
96    let (cohort_table, cohort_polars) = gen_cohort_table(sf, &mut mpc_exec_args)?;
97    let (diagnosis_table, diagnosis_polars) = gen_diagnosis_comorbidity_table(sf, &mut mpc_exec_args)?;
98
99    tracing::info!("Cohort table generated with {} rows", cohort_table.num_rows());
100    tracing::info!("Diagnosis table generated with {} rows", diagnosis_table.num_rows());
101
102
103    tracing::info!("Comorbidity query start");
104    let tot_start = Instant::now();
105
106
107    tracing::info!("Semi-join: diagnosis WHERE pid IN cohort");
108    let mut final_table = cohort_table.inner_join("pid", "pid", &diagnosis_table, &mut mpc_exec_args)?;
109
110
111    tracing::info!("GROUP BY diag, COUNT(*)");
112    let (e, perm, _) = final_table.group_by(vec!["diag"], &mut mpc_exec_args)?;
113    let _ = final_table.agg_count("cnt", &e, &perm, &mut mpc_exec_args)?;
114
115
116    tracing::info!("ORDER BY cnt DESC");
117    let _ = final_table.order_by("cnt", false, &mut mpc_exec_args)?;
118
119
120    tracing::info!("Comorbidity query execution completed");
121
122    if party_id == PartyID::ID0 {
123        tracing::info!("Total execution time: {:?}", tot_start.elapsed());
124    }
125    print_communication_stats(&mpc_exec_args, "Comorbidity");
126
127
128
129
130//************* polars verification *************//
131
132    
133    let mut result_table = final_table.project(vec!["diag", "cnt", "valid"])?;
134    let _ = result_table.order_by("valid", false, &mut mpc_exec_args);
135    
136    let sum_valid = result_table["valid"].prefix_sum();
137    let open_valid = open(sum_valid, mpc_exec_args.nets[0])?.0;
138    
139    result_table.head(open_valid as usize);
140
141    let mpc_result = result_table.open(&mut mpc_exec_args)?;
142
143    // Verification with Polars
144    if party_id == PartyID::ID0 {
145        tracing::info!("Comorbidity polars verification:");
146        let cohort = cohort_polars.unwrap();
147        let diagnosis = diagnosis_polars.unwrap();
148
149        // SQL: SELECT diag, COUNT(*) cnt FROM diagnosis WHERE pid IN cohort GROUP BY diag ORDER BY cnt DESC LIMIT 10
150        let comorbidity_result = diagnosis.lazy()
151            .join(
152                cohort.lazy(),
153                [col("pid")],
154                [col("pid")],
155                JoinType::Inner.into()
156            )
157            .group_by([col("diag")])
158            .agg([len().alias("cnt")])
159            .sort(
160                ["cnt", "diag"],
161                SortMultipleOptions::default().with_maintain_order(true).with_order_descending_multi([true, false])
162            )
163            .collect()?;
164
165        let mpc_diag = mpc_result["diag"].get_data();
166        let mpc_cnt = mpc_result["cnt"].get_data();
167
168        let polars_diag = comorbidity_result.column("diag")?.u64()?.into_no_null_iter().collect::<Vec<_>>();
169        let polars_cnt = comorbidity_result.column("cnt")?.cast(&DataType::UInt64)?.u64()?.into_no_null_iter().collect::<Vec<_>>();
170
171
172        assert_eq!(open_valid as usize, comorbidity_result.height(), "Result row count mismatch");
173        assert_eq!(mpc_cnt, &polars_cnt, "cnt mismatch");
174        assert_eq!(mpc_diag, &polars_diag, "diag mismatch");
175
176        tracing::info!("Verification passed!");
177    }
178
179    Ok(())
180}
