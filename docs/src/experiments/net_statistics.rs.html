60pub fn install_tracing() {
61    
62    let fmt_layer = fmt::layer()
63        .with_target(false)
64        .with_line_number(false)
65        .with_span_events(FmtSpan::CLOSE | FmtSpan::ENTER);
66    let filter_layer = EnvFilter::try_from_default_env()
67        .or_else(|_| EnvFilter::try_new("info"))
68        .unwrap();
69
70    tracing_subscriber::registry()
71        .with(filter_layer)
72        .with(fmt_layer)
73        .init();
74}
