17/// Performs a bitwise XOR operation on two shared values.
18pub fn xor<T: IntRing2k>(a: &RingShare<T>, b: &RingShare<T>) -> RingShare<T> {
19    a ^ b
20}

22/// Performs a bitwise XOR operation on a shared value and a public value.
23pub fn xor_public<T: IntRing2k>(
24    shared: &RingShare<T>,
25    public: &RingElement<T>,
26    id: PartyID,
27) -> RingShare<T> {
28    let mut res = shared.to_owned();
29    match id {
30        PartyID::ID0 => res.a ^= public,
31        PartyID::ID1 => res.b ^= public,
32        PartyID::ID2 => {}
33    }
34    res
35}

37/// Performs a bitwise OR operation on two shared values.
38pub fn or<T: IntRing2k, N: Network>(
39    a: &RingShare<T>,
40    b: &RingShare<T>,
41    net: &N,
42    state: &mut Rep3State,
43) -> eyre::Result<RingShare<T>>
44where
45    Standard: Distribution<T>,
46{
47    let xor = a ^ b;
48    let and = and(a, b, net, state)?;
49    Ok(xor ^ and)
50}

52/// Performs a bitwise OR operation on two vector of shared values.
53pub fn or_vec<T: IntRing2k, N: Network>(
54    a: &[RingShare<T>],
55    b: &[RingShare<T>],
56    net: &N,
57    state: &mut Rep3State,
58) -> eyre::Result<Vec<RingShare<T>>>
59where
60    Standard: Distribution<T>,
61{
62    let xor = izip!(a, b)
63        .map(|(a, b)| a ^ b)
64        .collect_vec();
65    let and = and_vec(a, b, net, state)?;
66    Ok(izip!(xor.iter(), and.iter())
67        .map(|(x, y)| *x ^ *y)
68        .collect_vec())
69}

71/// Performs a bitwise OR operation on a shared value and a public value.
72pub fn or_public<T: IntRing2k>(
73    shared: &RingShare<T>,
74    public: &RingElement<T>,
75    id: PartyID,
76) -> RingShare<T> {
77    let tmp = shared & public;
78    let xor = xor_public(shared, public, id);
79    xor ^ tmp
80}

82/// Performs a bitwise AND operation on two shared values.
83pub fn and<T: IntRing2k, N: Network>(
84    a: &RingShare<T>,
85    b: &RingShare<T>,
86    net: &N,
87    state: &mut Rep3State,
88) -> eyre::Result<RingShare<T>>
89where
90    Standard: Distribution<T>,
91{
92    let (mut mask, mask_b) = state.rngs.rand.random_elements::<RingElement<T>>();
93    mask ^= mask_b;
94    let local_a = (a & b) ^ mask;
95    let local_b = net.reshare(local_a)?;
96    Ok(RingShare::new_ring(local_a, local_b))
97}

177/// Shifts a share by a public value `F` to the right.
178///
179/// # Panics
180/// This method panics if `public` is larger than the of bits of
181/// the underlying `PrimeField`'s modulus'.
182pub fn shift_r_public<T: IntRing2k>(shared: &RingShare<T>, public: RingElement<T>) -> RingShare<T> {
183    // some special casing
184    if public.is_zero() {
185        return shared.to_owned();
186    }
187    let shift: usize = public
188        .0
189        .try_into()
190        .expect("can cast shift operand to usize");
191    shared >> shift
192}

199pub fn shift_l_public<T: IntRing2k>(shared: &RingShare<T>, public: RingElement<T>) -> RingShare<T> {
200    // some special casing

208    shared << shift
209}

211/// Performs the opening of a shared value and returns the equivalent public value.
212pub fn open<T: IntRing2k, N: Network>(a: &RingShare<T>, net: &N) -> eyre::Result<RingElement<T>> {
213    let c = net.reshare(a.b)?;
214    Ok(a.a ^ a.b ^ c)
215}
216/// Performs the opening of a vector of shared values and returns the equivalent public values.
217pub fn open_vec<T: IntRing2k, N: Network>(
218    a: &[RingShare<T>],
219    net: &N,
220) -> eyre::Result<Vec<RingElement<T>>> {
221    let c = net.reshare(a.iter().map(|x| x.b.clone()).collect_vec())?;
222    Ok(izip!(a, c)
223        .map(|(x, c)| x.a ^ x.b ^ c)
224        .collect()
225    )
226}

228/// Transforms a public value into a shared value: \[a\] = a.
229pub fn promote_to_trivial_share<T: IntRing2k>(
230    id: PartyID,
231    public_value: &RingElement<T>,
232) -> RingShare<T> {
233    match id {
234        PartyID::ID0 => RingShare::new_ring(public_value.to_owned(), RingElement::zero()),
235        PartyID::ID1 => RingShare::new_ring(RingElement::zero(), public_value.to_owned()),
236        PartyID::ID2 => RingShare::zero_share(),
237    }
238}

240/// Computes a CMUX: If `c` is `1`, returns `x_t`, otherwise returns `x_f`.
241pub fn cmux<T: IntRing2k, N: Network>(
242    c: &RingShare<T>,
243    x_t: &RingShare<T>,
244    x_f: &RingShare<T>,
245    net: &N,
246    state: &mut Rep3State,
247) -> eyre::Result<RingShare<T>>
248where
249    Standard: Distribution<T>,
250{
251    let xor = x_f ^ x_t;
252    let mut and = and(c, &xor, net, state)?;
253    and ^= x_f;
254    Ok(and)
255}

257/// Computes a CMUX for multiple inputs.
258pub fn cmux_many<T: IntRing2k, N: Network>(
259    c: &[RingShare<T>],
260    x_t: &[RingShare<T>],
261    x_f: &[RingShare<T>],
262    net: &N,
263    state: &mut Rep3State,
264) -> eyre::Result<Vec<RingShare<T>>>
265where
266    Standard: Distribution<T>,
267{
268    let xor_vec = izip!(x_t, x_f)
269        .map(|(x_t, x_f)| x_t ^ x_f)
270        .collect::<Vec<_>>();
271
272    let and = and_vec(c, &xor_vec, net, state)?;
273    Ok(izip!(and.iter(), x_f.iter())
274        .map(|(and, x_f)| *and ^ *x_f)
275        .collect::<Vec<_>>())
276}

282/// Compute a OR tree of the input vec
283pub fn or_tree<T: IntRing2k, N: Network>(
284    mut inputs: Vec<RingShare<T>>,
285    net: &N,
286    state: &mut Rep3State,
287) -> eyre::Result<RingShare<T>>
288where
289    Standard: Distribution<T>,
290{
291    let mut num = inputs.len();
292
293    tracing::debug!("starting or tree over {} elements", inputs.len());
294    while num > 1 {
295        tracing::trace!("binary tree still has {} elements", num);
296        let mod_ = num & 1;
297        num >>= 1;
298
299        let (a_vec, tmp) = inputs.split_at(num);
300        let (b_vec, leftover) = tmp.split_at(num);
301
302        let mut res = Vec::with_capacity(num);
303        // TODO WE WANT THIS BATCHED!!!
304        // THIS IS SUPER BAD
305        for (a, b) in izip!(a_vec.iter(), b_vec.iter()) {
306            res.push(or(a, b, net, state)?);
307        }
308
309        res.extend_from_slice(leftover);
310        inputs = res;
311
312        num += mod_;
313    }
314    let result = inputs[0];
315    tracing::debug!("we did it!");
316    Ok(result)
317}

319/// Computes a binary circuit to check whether the replicated binary-shared input x is zero or not. The output is a binary sharing of one bit.
320pub fn is_zero<T: IntRing2k, N: Network>(
321    x: &RingShare<T>,
322    net: &N,
323    state: &mut Rep3State,
324) -> eyre::Result<RingShare<Bit>>
325where
326    Standard: Distribution<T>,
327{
328    // negate
329    let mut x = !x;
330
331    // do ands in a tree
332    // TODO: Make and tree more communication efficient, ATM we send the full element for each level, even though they halve in size
333    let mut len = T::K;
334    debug_assert!(len.is_power_of_two());
335    while len > 1 {
336        // if len % 2 == 1 // Does not happen, we are in a ring with 2^k
337        len >>= 1;
338        let mask = (RingElement::one() << len) - RingElement::one();
339        let y = x >> len;
340        x = and(&(x & mask), &(y & mask), net, state)?;
341    }
342    // extract LSB
343    Ok(RingShare {
344        a: RingElement(Bit::new((x.a & RingElement::one()) == RingElement::one())),
345        b: RingElement(Bit::new((x.b & RingElement::one()) == RingElement::one())),
346    })
347}
