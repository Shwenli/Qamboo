1/* 
2* select
3 *      l_shipmode,
4 *      sum(case
5 *          when o_orderpriority ='1-URGENT'
6 *          or o_orderpriority ='2-HIGH'
7 *          then 1
8 *          else 0
9 *      end) as high_line_count,
10 *      sum(case
11 *          when o_orderpriority <> '1-URGENT'
12 *          and o_orderpriority <> '2-HIGH'
13 *          then 1
14 *          else 0
15 *      end) as low_line_count
16 *  from
17 *      orders,
18 *      lineitem
19 *  where
20 *      o_orderkey = l_orderkey
21 *      and l_shipmode in ('[SHIPMODE1]', '[SHIPMODE2]')
22 *      and l_commitdate < l_receiptdate
23 *      and l_shipdate < l_commitdate
24 *      and l_receiptdate >= date '[DATE]'
25 *      and l_receiptdate < date '[DATE]' + interval '1' year
26 *  group by
27 *      l_shipmode
28 *  order by
29 *      l_shipmode;

57const HIGH: u64 = 2;
58const URGENT: u64 = 1;
59
60const DATE: u64 = 30;
61const DATEANDINTERVAL: u64 = 80;
62const SHIPMODE1: u64 = 3;
63const SHIPMODE2: u64 = 4;
64
65
66#[derive(Parser)]
67struct Args {
68    /// The config file path
69    #[clap(short = 'c', long, value_name = "CONFIGDIR")]
70    config_dir: PathBuf,
71    /// The party ID (0, 1, 2)
72    #[clap(short = 'p', long, value_name = "PARTYID", default_value = "0")]
73    party_id: String,
74
75    /// The number of threads
76    #[clap(short = 't', long, value_name = "THREADS", default_value = "6")]
77    threads: usize,
78
79    /// Scale factor
80    #[clap(short = 's', long, value_name = "SF", default_value = "0.1")]
81    sf: f32,
82}

85fn main() -> Result<()> {
86    let args = Args::parse();
87    install_tracing();
88
89    let sf = args.sf; // scale factor for testing
90    let partyid=args.party_id.clone();
91    let default_threads = rayon::current_num_threads();
92
93    tracing::info!("setting up network");
94    let mut nets: Vec<TcpNetwork> = Vec::new();
95    let mut states: Vec<Rep3State> = Vec::new();
96
97    for i in 0..args.threads {
98        let file_path = PathBuf::from(format!("{}{}/config_party{}.toml", args.config_dir.display(), i, partyid));
99        let config: NetworkConfig =toml::from_str(&std::fs::read_to_string(file_path).context("opening config file")?).context("parsing config file")?;
100        let net = TcpNetwork::new(config)?;
101        let state = Rep3State::new(&net)?;
102        nets.push(net);
103        states.push(state);
104    }
105
106    if states.len() <= default_threads {
107        let diff = default_threads - states.len();
108        for _i in 0..diff{
109            let state = states[0].fork(0)?;
110            states.push(state);
111        }
112    }
113    
114    let nets = nets.iter().collect::<Vec<&TcpNetwork>>();
115    let mut states = states.iter_mut().collect::<Vec<&mut Rep3State>>();
116    let party_id = states[0].id;
117
118    let mut mpc_exec_args = NetStateArgs::new(
119        &nets,
120        &mut states,
121    );
122
123    tracing::info!("Network setup completed");
124    
125
126    tracing::info!("Generating tables");
127
128    let (mut lineitem_table, _lineitem_table_polars) = tpch_database_gen::gen_lineitem_table(sf, &mut mpc_exec_args)?;
129    tracing::info!("Lineitem table generated with {} rows", lineitem_table.num_rows());
130
131    let (mut orders_table, _orders_table_polars) = tpch_database_gen::gen_orders_table(sf, &mut mpc_exec_args)?;
132    tracing::info!("Orders table generated with {} rows", orders_table.num_rows());
133
134    tracing::info!("Generate tables completed");
135
136
137    tracing::info!("converting some columns to binary");
138
139    let l_shipmode_binary = lineitem_table["l_shipmode"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
140    lineitem_table.insert_column(l_shipmode_binary.get_name().to_string(),l_shipmode_binary);
141
142    let l_shipdate_binary = lineitem_table["l_shipdate"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
143    lineitem_table.insert_column(l_shipdate_binary.get_name().to_string(),l_shipdate_binary);
144
145    let l_commitdate_binary = lineitem_table["l_commitdate"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
146    lineitem_table.insert_column(l_commitdate_binary.get_name().to_string(),l_commitdate_binary);
147
148    let l_receiptdate_binary = lineitem_table["l_receiptdate"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
149    lineitem_table.insert_column(l_receiptdate_binary.get_name().to_string(),l_receiptdate_binary);
150
151    let o_orderpriority_binary = orders_table["o_orderpriority"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
152    orders_table.insert_column(o_orderpriority_binary.get_name().to_string(),o_orderpriority_binary);
153
154
155    tracing::info!("Projecting tables");
156    
157    let l_col_names = vec!["l_shipmode", "[l_shipmode]", "[l_shipdate]", "[l_commitdate]", "l_orderkey", "[l_receiptdate]", "valid"];
158    let mut lineitem_table = lineitem_table.project(l_col_names)?;
159
160    let o_col_names = vec!["[o_orderpriority]", "o_orderkey", "valid"];
161    let mut orders_table = orders_table.project(o_col_names)?;
162
163    tracing::info!("Projection completed");
164
165
166    tracing::info!("Q12 start");
167    let tot_start = Instant::now();
168
169    tracing::info!("case when o_orderpriority ='1-URGENT or o_orderpriority ='2-HIGH' then 1 else 0");
170
171    let f_values_high = vec![URGENT, HIGH];
172
173    let high_line_data = get_one_share_vec::<u64>(orders_table["[o_orderpriority]"].len(), party_id)?;
174    let mut high_line_column = ShareColumn::<Rep3RingShare<u64>>::new(
175        high_line_data,
176        ShareType::Arithmetic,
177        "highline".to_string(),
178    );
179
180    let eq_f = orders_table["[o_orderpriority]"].in_public_binary(&f_values_high, &mut mpc_exec_args)?;
181    high_line_column *= (&eq_f, &mut mpc_exec_args);
182
183    orders_table.insert_column(high_line_column.get_name().to_string(), high_line_column);
184    orders_table.delete_column("[o_orderpriority]");
185
186
187    tracing::info!("l_shipmode in ('[SHIPMODE1]', '[SHIPMODE2]')");
188
189    let c1 = lineitem_table["[l_shipmode]"].eq_public_binary(&SHIPMODE1, &mut mpc_exec_args)?;
190    let c2 = lineitem_table["[l_shipmode]"].eq_public_binary(&SHIPMODE2, &mut mpc_exec_args)?;
191    let s1 = c1.or(&c2, &mut mpc_exec_args)?;
192
193    tracing::info!("l_commitdate < l_receiptdate and l_shipdate < l_commitdate");
194
195    let c1 = lineitem_table["[l_commitdate]"].lt_binary(&lineitem_table["[l_receiptdate]"], &mut mpc_exec_args)?;
196    let c2 = lineitem_table["[l_shipdate]"].lt_binary(&lineitem_table["[l_commitdate]"], &mut mpc_exec_args)?;
197
198    let s2 = c1.and(&c2, &mut mpc_exec_args)?;
199
200    tracing::info!("l_receiptdate >= date '[DATE]' and l_receiptdate < date '[DATE]' + interval '1' year");
201
202    let c1 = lineitem_table["[l_receiptdate]"].ge_public_binary(&DATE, &mut mpc_exec_args)?;
203    let c2 = lineitem_table["[l_receiptdate]"].lt_public_binary(&DATEANDINTERVAL, &mut mpc_exec_args)?;
204
205    let s3 = c1.and(&c2, &mut mpc_exec_args)?;
206
207    let s = s1.and(&s2, &mut mpc_exec_args)?.and(&s3, &mut mpc_exec_args)?;
208    let _ = lineitem_table.filter_directed_by_bool(s.get_data(), &mut mpc_exec_args)?;
209
210
211    let lineitem_table = lineitem_table.project(vec!["l_shipmode", "l_orderkey", "valid"])?;
212
213
214    tracing::info!("o_orderkey = l_orderkey");
215
216    let mut final_table = orders_table.inner_join(
217        "o_orderkey", 
218        "l_orderkey",
219        &lineitem_table, 
220        &mut mpc_exec_args
221    )?;
222
223
224    tracing::info!("group by l_shipmode");
225
226    let group_by_cols = vec!["l_shipmode"];
227    
228    let (e,perm,_) = final_table.group_by(
229        group_by_cols,
230        &mut mpc_exec_args,
231    )?;
232
233
234    tracing::info!("sum(highline)");
235
236    let to_agg_name = "highline";
237    let new_agg_name = "high_line_count";
238
239    let _ = final_table.agg_sum(
240        to_agg_name,
241        new_agg_name,
242        &e,
243        &perm,
244        &mut mpc_exec_args,
245    )?;
246
247
248    tracing::info!("Q12 execution completed");
249
250    if party_id == PartyID::ID0 {
251        tracing::info!("Total Q12 execution time: {:?}", tot_start.elapsed());
252    }
253    print_communication_stats(&mpc_exec_args, "Q12");
254    
255
256
257//************* polars verification *************//
258
259    let mut result_table = final_table.project(vec!["l_shipmode", "high_line_count", "valid"])?;
260    // sort valid to top
261    let _ = result_table.order_by("valid", false, &mut mpc_exec_args)?;
262    
263    let sum_valid = result_table["valid"].prefix_sum();
264    let open_valid = open(sum_valid, mpc_exec_args.nets[0])?.0;
265    tracing::info!("open valid: {}", open_valid);
266    result_table.head(open_valid as usize);
267
268    let mpc_result = result_table.open(&mut mpc_exec_args)?;
269
270    if party_id == PartyID::ID0 {
271        tracing::info!("Q12 polars:");
272        let lineitem = _lineitem_table_polars.unwrap();
273        let orders = _orders_table_polars.unwrap();
274
275        let final_df = orders.lazy()
276            .join(
277                lineitem.lazy(),
278                [col("o_orderkey")],
279                [col("l_orderkey")],
280                JoinArgs::new(JoinType::Inner)
281            )
282            .filter(col("l_shipmode").eq(lit(SHIPMODE1)).or(col("l_shipmode").eq(lit(SHIPMODE2))))
283            .filter(col("l_commitdate").lt(col("l_receiptdate")))
284            .filter(col("l_shipdate").lt(col("l_commitdate")))
285            .filter(
286                col("l_receiptdate").gt_eq(lit(DATE))
287                .and(col("l_receiptdate").lt(lit(DATEANDINTERVAL)))
288            )
289            .with_columns(vec![
290                col("o_orderpriority").eq(lit(URGENT)).or(col("o_orderpriority").eq(lit(HIGH))).alias("line_count")
291            ])
292            .group_by(vec![col("l_shipmode")])
293            .agg(vec![
294                col("line_count").sum().alias("high_line_count")
295            ])
296            .sort(["l_shipmode"], SortMultipleOptions::default().with_maintain_order(true))
297            .collect()?;
298        
299        tracing::info!("Polars result: {:?}", final_df.height());
300        
301        let mpc_high_line_count = mpc_result["high_line_count"].get_data();
302        let mpc_l_shipmode = mpc_result["l_shipmode"].get_data();
303
304        let polars_high_line_count = final_df.column("high_line_count")?.cast(&polars::datatypes::DataType::UInt64)?.u64()?.into_no_null_iter().collect::<Vec<_>>();
305        let polars_l_shipmode = final_df.column("l_shipmode")?.u64()?.into_no_null_iter().collect::<Vec<_>>();
306
307        assert_eq!(mpc_high_line_count, polars_high_line_count);
308        assert_eq!(mpc_l_shipmode, polars_l_shipmode);
309        
310        
311        tracing::info!("Verification passed!");
312    }
313    
314    Ok(())
315}
