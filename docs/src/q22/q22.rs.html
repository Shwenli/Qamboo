63const I1: u64 = 13;
64const I2: u64 = 31;
65const I3: u64 = 23;
66const I4: u64 = 29;
67const I5: u64 = 30;
68const I6: u64 = 18;
69const I7: u64 = 17;

71
72#[derive(Parser)]
73struct Args {
74    /// The config file path
75    #[clap(short = 'c', long, value_name = "CONFIGDIR")]
76    config_dir: PathBuf,
77
78    /// The party ID (0, 1, 2)
79    #[clap(short = 'p', long, value_name = "PARTYID", default_value = "0")]
80    party_id: String,
81
82    /// The number of threads
83    #[clap(short = 't', long, value_name = "THREADS", default_value = "6")]
84    threads: usize,
85
86    /// Scale factor
87    #[clap(short = 's', long, value_name = "SF", default_value = "0.1")]
88    sf: f32,
89}
90
91
92fn main() -> Result<()> {
93    let args = Args::parse();
94    install_tracing();
95
96    let sf = args.sf; // scale factor for testing
97    let partyid=args.party_id.clone();
98    let default_threads = rayon::current_num_threads();
99
100    tracing::info!("setting up network");
101    let mut nets: Vec<TcpNetwork> = Vec::new();
102    let mut states: Vec<Rep3State> = Vec::new();
103
104    for i in 0..args.threads {
105        let file_path = PathBuf::from(format!("{}{}/config_party{}.toml", args.config_dir.display(), i, partyid));
106        let config: NetworkConfig =toml::from_str(&std::fs::read_to_string(file_path).context("opening config file")?).context("parsing config file")?;
107        let net = TcpNetwork::new(config)?;
108        let state = Rep3State::new(&net)?;
109        nets.push(net);
110        states.push(state);
111    }
112
113    if states.len() <= default_threads {
114        let diff = default_threads - states.len();
115        for _i in 0..diff{
116            let state = states[0].fork(0)?;
117            states.push(state);
118        }
119    }
120    
121    let nets = nets.iter().collect::<Vec<&TcpNetwork>>();
122    let mut states = states.iter_mut().collect::<Vec<&mut Rep3State>>();
123    let party_id = states[0].id;
124
125    let mut mpc_exec_args = NetStateArgs::new(
126        &nets,
127        &mut states,
128    );
129
130    tracing::info!("Network setup completed");
131    
132
133    tracing::info!("Generating tables");
134
135    let (mut customer_table, customer_table_polars) = tpch_database_gen::gen_customer_table(sf, &mut mpc_exec_args)?;
136    tracing::info!("Customer table generated with {} rows", customer_table.num_rows());
137
138    let (orders_table, orders_table_polars) = tpch_database_gen::gen_orders_table(sf, &mut mpc_exec_args)?;
139    tracing::info!("Orders table generated with {} rows", orders_table.num_rows());
140
141
142    tracing::info!("converting some columns to binary");
143
144    let c_cntrycode_binary = customer_table["c_cntrycode"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
145    customer_table.insert_column("[c_cntrycode]".to_string(), c_cntrycode_binary);
146
147    let c_acctbal_binary = customer_table["c_acctbal"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
148    customer_table.insert_column("[c_acctbal]".to_string(), c_acctbal_binary);
149    
150
151    tracing::info!("Projecting tables");
152
153    let o_col_names = vec!["o_custkey", "valid"];
154    let orders_table = orders_table.project(o_col_names)?;
155
156    let c_col_names = vec!["c_custkey", "c_acctbal", "[c_cntrycode]", "c_cntrycode", "[c_acctbal]", "valid"];
157    let mut customer_table = customer_table.project(c_col_names)?;
158
159
160    tracing::info!("Q22 start");
161    let tot_start = Instant::now();
162
163    tracing::info!("substring(c_phone from 1 for 2) in ('[I1]','[I2]','[I3]','[I4]','[I5]','[I6]','[I7]')");
164
165    let pub_elements = vec![I1, I2, I3, I4, I5, I6, I7];
166    let res_in = customer_table["[c_cntrycode]"].in_public_binary(&pub_elements, &mut mpc_exec_args)?;
167    let _ = customer_table.filter_directed_by_bool(res_in.get_data(), &mut mpc_exec_args)?;
168    customer_table.delete_column("[c_cntrycode]");
169
170    let _ = customer_table.filter_public("[c_acctbal]",  Predicate::GreaterThanBinary, &0, &mut mpc_exec_args)?;
171    customer_table.delete_column("[c_acctbal]");
172    
173    /* 
174    let sum_valid = customer_table["valid"].prefix_sum();
175    let open_valid = open(sum_valid, &net0)?.0;
176    tracing::info!("Rows after filter: {}", open_valid);
177    */
178
179    tracing::info!("select avg(c_acctbal)");
180    let valid_clone = customer_table["valid"].clone();
181    customer_table["c_acctbal"] *= (&valid_clone, &mut mpc_exec_args);
182    let sum_acctbal = customer_table["c_acctbal"].prefix_sum();
183    let sum_valid = customer_table["valid"].prefix_sum();
184    let avg_acctbal = div(&sum_acctbal, &sum_valid, 32 as usize, mpc_exec_args.nets[0], mpc_exec_args.states[0])?;
185
186    let avg_acctbal_col_data = vec![avg_acctbal; customer_table.num_rows()];
187    let avg_acctbal_col = ShareColumn::new(
188        avg_acctbal_col_data,
189        ShareType::Arithmetic,
190        "avg_acctbal".to_string(),
191    );
192    customer_table.insert_column("avg_acctbal".to_string(), avg_acctbal_col);
193
194
195    tracing::info!("c_acctbal > avg_acctbal");
196
197    let _ = customer_table.filter_shared("c_acctbal",  "avg_acctbal", Predicate::GreaterThan,  &mut mpc_exec_args)?;
198    customer_table.delete_column("avg_acctbal");
199
200    /* 
201    let sum_valid = customer_table["valid"].prefix_sum();
202    let open_valid = open(sum_valid, &net0)?.0;
203    tracing::info!("Rows after filter: {}", open_valid);
204    */
205
206
207    tracing::info!("not exists o_custkey = c_custkey");
208
209    let _ = customer_table.anti_join(
210        "c_custkey",
211        "o_custkey",
212        &orders_table,
213        &mut mpc_exec_args,
214    )?;
215
216
217    tracing::info!("group by cntrycode");
218    
219    let (e,perm,_) = customer_table.group_by(
220            vec!["c_cntrycode"],
221        &mut mpc_exec_args,
222    )?;
223
224
225    tracing::info!("count(*) as numcust, sum(c_acctbal) as totacctbal");
226
227    customer_table.agg_count(
228        "numcust",
229        &e,
230        &perm,
231        &mut mpc_exec_args,
232    )?;
233   
234    customer_table.agg_sum(
235        "c_acctbal",
236        "totacctbal",
237        &e,
238        &perm,
239        &mut mpc_exec_args,
240    )?;
241
242
243    tracing::info!("Q22 execution completed");
244
245    if party_id == PartyID::ID0 {
246        tracing::info!("Total Q22 execution time: {:?}", tot_start.elapsed());
247    }
248    print_communication_stats(&mpc_exec_args, "Q22");
249    
250    
251    
252//************* polars verification *************//
253
254    let mut final_table = customer_table;
255
256    let sum_valid = final_table["valid"].prefix_sum();
257    let open_valid = open(sum_valid, mpc_exec_args.nets[0])?.0;
258    tracing::info!("open valid: {}", open_valid);
259    final_table.head(open_valid as usize);
260
261    let mpc_result = final_table.open(&mut mpc_exec_args)?;
262
263    
264    if party_id == PartyID::ID0 {
265        tracing::info!("Q22 polars:");
266
267        let orders = orders_table_polars.unwrap();
268        let customer = customer_table_polars.unwrap();
269
270        // q1 = customer.filter(substring(c_phone, 1, 2) in country_codes)
271        let q1 = customer.clone().lazy()
272            .filter(
273                col("c_cntrycode").eq(lit(I1))
274                .or(col("c_cntrycode").eq(lit(I2)))
275                .or(col("c_cntrycode").eq(lit(I3)))
276                .or(col("c_cntrycode").eq(lit(I4)))
277                .or(col("c_cntrycode").eq(lit(I5)))
278                .or(col("c_cntrycode").eq(lit(I6)))
279                .or(col("c_cntrycode").eq(lit(I7)))
280            )
281            .filter(col("c_acctbal").gt(lit(0)))
282            .select([col("c_acctbal"), col("c_custkey"), col("c_cntrycode")]);
283        
284        tracing::info!("q1 rows: {}", q1.clone().collect()?.height());
285
286        // q2 = avg(c_acctbal) where c_acctbal > 0
287        let q2 = q1.clone().lazy()
288            .with_columns([
289                col("c_acctbal").sum().alias("sum_acctbal"),
290                col("c_acctbal").count().alias("count_acctbal")
291            ])
292            .with_column(
293                (col("sum_acctbal") / col("count_acctbal")).alias("avg_acctbal")
294            )
295            .filter(col("c_acctbal").gt(col("avg_acctbal")))
296            .select([col("c_cntrycode"), col("c_acctbal"), col("c_custkey")]);        
297        
298        tracing::info!("q2 rows: {}", q2.clone().collect()?.height());
299
300        let q22_polars = q2.join(
301                orders.lazy(),
302                [col("c_custkey")],
303                [col("o_custkey")],
304                JoinArgs::new(JoinType::Anti)
305            )
306            .group_by([col("c_cntrycode")])
307            .agg([
308                col("c_cntrycode").count().alias("numcust"),
309                col("c_acctbal").sum().alias("totacctbal")
310            ])
311            .sort(
312                ["c_cntrycode"],
313                SortMultipleOptions::default().with_maintain_order(true)
314            )
315            .collect()?;
316        
317
318        let mpc_cntrycode = mpc_result["c_cntrycode"].get_data();
319        let mpc_numcust = mpc_result["numcust"].get_data();
320        let mpc_totacctbal = mpc_result["totacctbal"].get_data();
321
322        let polars_cntrycode = q22_polars.column("c_cntrycode")?.cast(&DataType::UInt64)?.u64()?.into_no_null_iter().collect::<Vec<_>>();
323        let polars_numcust = q22_polars.column("numcust")?.cast(&DataType::UInt64)?.u64()?.into_no_null_iter().collect::<Vec<_>>();
324        let polars_totacctbal = q22_polars.column("totacctbal")?.cast(&DataType::UInt64)?.u64()?.into_no_null_iter().collect::<Vec<_>>();
325        tracing::info!("rows of polars: {:?}", q22_polars.height());
326        // tracing::info!("MPC cntrycode: {:?}", mpc_cntrycode);
327        // tracing::info!("Polars cntrycode: {:?}", polars_cntrycode);
328        
329        assert_eq!(mpc_cntrycode, &polars_cntrycode, "cntrycode mismatch");
330        assert_eq!(mpc_numcust, &polars_numcust, "numcust mismatch");
331        assert_eq!(mpc_totacctbal, &polars_totacctbal, "totacctbal mismatch");
332        tracing::info!("Passed! Q22 result matched with Polars.");
333    }
334    
335    Ok(())
