26use std::time::Instant;
27use std::path::PathBuf;
28use std::vec;
29use clap::Parser;
30use color_eyre::{Result, eyre::Context};
31use net::tcp::{TcpNetwork, NetworkConfig};
32use random::rep3::Rep3State;
33use random::MpcState;
34use communication::rep3::id::PartyID;
35use experiments::net_statistics::install_tracing;
36use experiments::net_statistics::print_communication_stats;
37use experiments::tpch_database_gen;
38use table::column_operator::TransformBetweenArithAndBinary;
39use table::column_operator::{ColumnBooleanOperator, PrefixSum};
40use table::table_operator::{Filter, Groupby, AggFunc, Join, Open, OrderBy, Project};
41use table::predicate::Predicate;
42use table::NetStateArgs;
43use protocols::rep3_ring::arithmetic::open;
44use polars::prelude::*;
45
46
47const DATE: u64 = 70;
48const DATE_INTERVAL: u64 = 30;
49
50
51#[derive(Parser)]
52struct Args {
53    /// The config file path
54    #[clap(short = 'c', long, value_name = "CONFIGDIR")]
55    config_dir: PathBuf,
56    
57    /// The party ID (0, 1, 2)
58    #[clap(short = 'p', long, value_name = "PARTYID", default_value = "0")]
59    party_id: String,
60
61    /// The number of threads
62    #[clap(short = 't', long, value_name = "THREADS", default_value = "6")]
63    threads: usize,
64
65    /// Scale factor
66    #[clap(short = 's', long, value_name = "SF", default_value = "0.1")]
67    sf: f32,
68}
69
70
71fn main() -> Result<()> {
72    let args = Args::parse();
73    install_tracing();
74
75    let sf = args.sf; // scale factor for testing
76    let partyid= args.party_id.clone();
77    let default_threads = rayon::current_num_threads() / 2;
78
79    tracing::info!("Q4 with SF: {}", sf);
80    
81    tracing::info!("setting up network");
82    let mut nets: Vec<TcpNetwork> = Vec::new();
83    let mut states: Vec<Rep3State> = Vec::new();
84
85    for i in 0..args.threads {
86        let file_path = PathBuf::from(format!("{}{}/config_party{}.toml", args.config_dir.display(), i, partyid));
87        let config: NetworkConfig =toml::from_str(&std::fs::read_to_string(file_path).context("opening config file")?).context("parsing config file")?;
88        let net = TcpNetwork::new(config)?;
89        let state = Rep3State::new(&net)?;
90        nets.push(net);
91        states.push(state);
92    }
93
94    if states.len() <= default_threads {
95        let diff = default_threads - states.len();
96        for _i in 0..diff{
97            let state = states[0].fork(0)?;
98            states.push(state);
99        }
100    }
101    
102    let nets = nets.iter().collect::<Vec<&TcpNetwork>>();
103    let mut states = states.iter_mut().collect::<Vec<&mut Rep3State>>();
104    let party_id = states[0].id;
105
106    let mut mpc_exec_args = NetStateArgs::new(
107        &nets,
108        &mut states,
109    );
110
111    tracing::info!("Network setup completed");
112    
113
114    let (mut lineitem_table, lineitem_table_polars) = tpch_database_gen::gen_lineitem_table(sf, &mut mpc_exec_args)?;
115    tracing::info!("Lineitem table generated with {} rows", lineitem_table.num_rows());
116
117    let (mut orders_table, orders_table_polars) = tpch_database_gen::gen_orders_table(sf, &mut mpc_exec_args)?;
118    tracing::info!("Orders table generated with {} rows", orders_table.num_rows());
119
120    let l_commidate_binary = lineitem_table["l_commitdate"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
121    lineitem_table.insert_column(l_commidate_binary.get_name().to_string(),l_commidate_binary);
122
123    let l_receiptdate_binary = lineitem_table["l_receiptdate"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
124    lineitem_table.insert_column(l_receiptdate_binary.get_name().to_string(),l_receiptdate_binary);
125
126    let o_orderdate_binary = orders_table["o_orderdate"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
127    orders_table.insert_column(o_orderdate_binary.get_name().to_string(),o_orderdate_binary);
128
129
130    tracing::info!("Projecting tables");
131
132    let l_col_names = vec!["l_orderkey", "[l_commitdate]", "[l_receiptdate]", "valid"];
133    let mut lineitem_table = lineitem_table.project(l_col_names)?;
134
135    let o_col_names = vec!["o_orderkey", "o_orderpriority", "[o_orderdate]", "valid"];
136    let mut orders_table = orders_table.project(o_col_names)?;
137    
138    tracing::info!("Projection completed");
139
140
141    tracing::info!("Q4 start");
142    let tot_start = Instant::now();
143
144    tracing::info!("Filtering tables: lineitem");
145
146    let l_lhs = "[l_commitdate]";
147    let l_rhs = "[l_receiptdate]";
148
149    let _ = lineitem_table.filter_shared(
150        l_lhs,
151        l_rhs,
152        Predicate::LessThanBinary,
153        &mut mpc_exec_args,
154    )?;
155
156    lineitem_table.delete_column("[l_commitdate]");
157    lineitem_table.delete_column("[l_receiptdate]");
158
159    tracing::info!("lineitem filter completed");
160
161
162    tracing::info!("o_orderdate >= date '[DATE]' and o_orderdate < date '[DATE]' + interval '3' month");
163    
164    let c1 = orders_table["[o_orderdate]"].ge_public_binary(&DATE, &mut mpc_exec_args)?;
165    let c2 = orders_table["[o_orderdate]"].lt_public_binary(&(DATE + DATE_INTERVAL), &mut mpc_exec_args)?;
166    let c = c1.and(&c2, &mut mpc_exec_args)?;
167    
168    orders_table.filter_directed_by_bool(c.get_data(), &mut mpc_exec_args)?;
169
170    orders_table.delete_column("[o_orderdate]");
171
172    
173    tracing::info!("semi join");
174
175    let k_l_name = "l_orderkey";
176    let k_r_name = "o_orderkey";
177
178    let _ = orders_table.semi_join(
179        k_r_name,
180        k_l_name,
181        &lineitem_table,
182        &mut mpc_exec_args,
183    )?;
184
185
186    tracing::info!("group by o_orderpriority");
187
188    let group_by_col_names = vec!["o_orderpriority"];
189
190    let (e,perm,_) = orders_table.group_by(
191        group_by_col_names,
192        &mut mpc_exec_args,
193    )?;
194
195
196    tracing::info!("count(*) as order_count");
197
198    let _ = orders_table.agg_count(
199        "order_count",
200        &e,
201        &perm,
202        &mut mpc_exec_args,
203    )?;
204
205    tracing::info!("Aggregate completed");
206
207
208    tracing::info!("order by o_orderpriority");
209
210    let _ = orders_table.order_by(
211        "o_orderpriority",
212        true,
213        &mut mpc_exec_args,
214    )?;
215
216
217    tracing::info!("Q4 execution completed");
218
219    if party_id == PartyID::ID0 {
220        tracing::info!("Total Q4 execution time: {:?}", tot_start.elapsed());
221    }
222    print_communication_stats(&mpc_exec_args, "Q4");
223
224
225
226
227//************* polars verification *************//
228
229    let _ = orders_table.order_by("valid", false, &mut mpc_exec_args)?;
230    let mut result_table = orders_table.project(vec!["o_orderpriority", "order_count", "valid"])?;
231
232    let sum_valid = result_table["valid"].prefix_sum();
233    let open_valid = open(sum_valid, mpc_exec_args.nets[0])?.0;
234    tracing::info!("open valid: {}", open_valid);
235    result_table.head(open_valid as usize);
236
237    let mpc_result = result_table.open(&mut mpc_exec_args)?;
238
239    if party_id == PartyID::ID0 {
240        tracing::info!("Q4 polars:");
241
242        let lineitem = lineitem_table_polars.unwrap();
243        let orders = orders_table_polars.unwrap();
244        
245        let q4_result = orders.lazy()
246            .filter(
247                col("o_orderdate").gt_eq(lit(DATE))
248                .and(col("o_orderdate").lt(lit(DATE + DATE_INTERVAL)))
249            )
250            .join(
251                lineitem.lazy().filter(
252                    col("l_commitdate").lt(col("l_receiptdate"))
253                ),
254                [col("o_orderkey")],
255                [col("l_orderkey")],
256                JoinArgs::new(JoinType::Semi)
257            )
258            .group_by([col("o_orderpriority")])
259            .agg([
260                len().alias("order_count")
261            ])
262            .sort(["o_orderpriority"], SortMultipleOptions::default())
263            .collect()
264            .unwrap();
265
266        let mpc_o_orderpriority = mpc_result["o_orderpriority"].get_data();
267        let mpc_order_count = mpc_result["order_count"].get_data();
268
269        let polars_o_orderpriority = q4_result.column("o_orderpriority")?.u64()?.into_no_null_iter().collect::<Vec<_>>();
270        let polars_order_count = q4_result.column("order_count")?.cast(&DataType::UInt64)?.u64()?.into_no_null_iter().collect::<Vec<_>>();
271
272        tracing::info!("rows of polars: {:?}", polars_o_orderpriority.len());
273        assert_eq!(mpc_o_orderpriority, &polars_o_orderpriority);
274        assert_eq!(mpc_order_count, &polars_order_count);
275
276        tracing::info!("Q4 Passed: Qamboo result MATCHES Polars result !");
277    }
278
279    Ok(())
280}
