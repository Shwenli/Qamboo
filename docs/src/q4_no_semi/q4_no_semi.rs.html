36use experiments::net_statistics::install_tracing;
37use experiments::net_statistics::print_communication_stats;

44#[derive(Parser)]
45struct Args {
46    /// The config file path
47    #[clap(short = 'c', long, value_name = "CONFIGDIR")]
48    config_dir: PathBuf,
49    /// The party ID (0, 1, 2)
50    #[clap(short = 'p', long, value_name = "PARTYID", default_value = "0")]
51    party_id: String,
52
53    /// The number of threads
54    #[clap(short = 't', long, value_name = "THREADS", default_value = "6")]
55    threads: usize,
56
57    /// Scale factor
58    #[clap(short = 's', long, value_name = "SF", default_value = "0.1")]
59    sf: f32,
60}
61
62
63fn main() -> Result<()> {
64    let args = Args::parse();
65    install_tracing();
66
67    let sf = args.sf; // scale factor for testing
68    let partyid= args.party_id.clone();
69    let default_threads = rayon::current_num_threads() / 2;
70    
71    tracing::info!("setting up network");
72    let mut nets: Vec<TcpNetwork> = Vec::new();
73    let mut states: Vec<Rep3State> = Vec::new();
74
75    for i in 0..args.threads {
76        let file_path = PathBuf::from(format!("{}{}/config_party{}.toml", args.config_dir.display(), i, partyid));
77        let config: NetworkConfig =toml::from_str(&std::fs::read_to_string(file_path).context("opening config file")?).context("parsing config file")?;
78        let net = TcpNetwork::new(config)?;
79        let state = Rep3State::new(&net)?;
80        nets.push(net);
81        states.push(state);
82    }
83
84    if states.len() <= default_threads {
85        let diff = default_threads - states.len();
86        for _i in 0..diff{
87            let state = states[0].fork(0)?;
88            states.push(state);
89        }
90    }
91    
92    let nets = nets.iter().collect::<Vec<&TcpNetwork>>();
93    let mut states = states.iter_mut().collect::<Vec<&mut Rep3State>>();
94    let party_id = states[0].id;
95
96    let mut mpc_exec_args = NetStateArgs::new(
97        &nets,
98        &mut states,
99    );
100
101    tracing::info!("Network setup completed");
102    
103
104    let (lineitem_table, _lineitem_table_polars) = tpch_database_gen::gen_lineitem_table(sf, &mut mpc_exec_args)?;
105    tracing::info!("Lineitem table generated with {} rows", lineitem_table.num_rows());
106
107    let (orders_table, _orders_table_polars) = tpch_database_gen::gen_orders_table(sf, &mut mpc_exec_args)?;
108    tracing::info!("Orders table generated with {} rows", orders_table.num_rows());
109
110    // TODO: Implement Q4 query logic using lineitem_table
111
112    //*LineItem.project({"[OrderKey]", "[ShipDate]", "ExtendedPrice", "Discount"});
113    //*Orders.project({"[OrderKey]", "[CustKey]", "[OrderDate]"});
114    //*Customers.project({"[CustKey]", "[MktSegment]"});
115
116    tracing::info!("Projecting tables");
117
118    let l_col_names = vec!["l_orderkey", "l_commitdate", "l_receiptdate", "valid"];
119    let mut lineitem_table = lineitem_table.project(l_col_names)?;
120
121    let o_col_names = vec!["o_orderkey", "o_orderdate", "o_orderpriority", "valid"];
122    let mut orders_table = orders_table.project(o_col_names)?;
123    
124    tracing::info!("Projection completed");
125
126
127    tracing::info!("Q4 start");
128    let tot_start = Instant::now();
129
130    tracing::info!("Filtering tables: lineitem");
131
132    let l_lhs = "l_commitdate";
133    let l_rhs = "l_receiptdate";
134
135    let _ = lineitem_table.filter_shared(
136        l_lhs,
137        l_rhs,
138        Predicate::LessThan,
139        &mut mpc_exec_args,
140    )?;
141
142    tracing::info!("lineitem filter completed");
143
144
145    tracing::info!("Filtering tables: orders");
146    let start = Instant::now();
147
148    let o_date = "o_orderdate";
149    let _ = orders_table.filter_public(
150        o_date,
151        Predicate::GreaterOrEqual,
152        &88u64,
153        &mut mpc_exec_args,
154    )?;
155
156    let o_date = "o_orderdate";
157    let _ = orders_table.filter_public(
158        o_date,
159        Predicate::LessOrEqual,
160        &94u64,
161        &mut mpc_exec_args,
162    )?;
163
164    tracing::info!("orderstable filter time: {:?}", start.elapsed());
165
166    
167    tracing::info!("inner join start");
168    let start = Instant::now();
169
170    let k_l_name = "l_orderkey";
171    let k_r_name = "o_orderkey";
172
173    let mut orderslineitem_table = orders_table.inner_join(
174        k_r_name,
175        k_l_name,
176        &lineitem_table,
177        &mut mpc_exec_args,
178    )?;
179
180    tracing::info!("inner join time: {:?}", start.elapsed());
181    
182
183    tracing::info!("table group by");
184    let start = Instant::now();
185
186    let group_by_col_names = vec!["o_orderpriority"];
187
188    let (e,perm,_) = orderslineitem_table.group_by(
189        group_by_col_names,
190        &mut mpc_exec_args,
191    )?;
192
193    tracing::info!("Group by time: {:?}", start.elapsed());
194    
195
196    tracing::info!("table aggregate");
197    let start = Instant::now();
198
199    let new_agg_name = "order_count";
200
201    let _ = orderslineitem_table.agg_count(
202        new_agg_name,
203        &e,
204        &perm,
205        &mut mpc_exec_args,
206    )?;
207
208    tracing::info!("Aggregate time: {:?}", start.elapsed());
209    tracing::info!("Aggregate completed");
210
211
212    tracing::info!("Q4 execution completed");
213
214    if party_id == PartyID::ID0 {
215        tracing::info!("Total Q4_no_semi execution time: {:?}", tot_start.elapsed());
216    }
217    print_communication_stats(&mpc_exec_args, "Q4_no_semi");
218
219    Ok(())
220}
