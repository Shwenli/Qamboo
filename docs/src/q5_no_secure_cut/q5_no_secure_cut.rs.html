1/*
2 * select
3 *     n_name,
4 *     sum(l_extendedprice * (1 - l_discount)) as revenue
5 * from
6 *     customer,
7 *     orders,
8 *     lineitem,
9 *     supplier,
10 *     nation,
11 *     region
12 * where
13 *     c_custkey = o_custkey
14 *     and l_orderkey = o_orderkey
15 *     and l_suppkey = s_suppkey
16 *     and c_nationkey = s_nationkey
17 *     and s_nationkey = n_nationkey
18 *     and n_regionkey = r_regionkey
19 *     and r_name = '[REGION]'
20 *     and o_orderdate >= date '[DATE]'
21 *     and o_orderdate < date '[DATE]' + interval '1' year
22 * group by n_name
23 * order by revenue desc;
24 */

49const DATE :u64 = 70;
50const DATE_INTERVAL :u64 = 30;  // Arbitrary date interval to account for date format
51const REGION :u64= 5;
52
53
54#[derive(Parser)]
55struct Args {
56    /// The config file path
57    #[clap(short = 'c', long, value_name = "CONFIGDIR")]
58    config_dir: PathBuf,
59    /// The party ID (0, 1, 2)
60    #[clap(short = 'p', long, value_name = "PARTYID", default_value = "0")]
61    party_id: String,
62
63    /// The number of threads
64    #[clap(short = 't', long, value_name = "THREADS", default_value = "6")]
65    threads: usize,
66
67    /// Scale factor
68    #[clap(short = 's', long, value_name = "SF", default_value = "0.1")]
69    sf: f32,
70}
71
72
73fn main() -> Result<()> {
74    let args = Args::parse();
75    install_tracing();
76
77    let sf = args.sf; // scale factor for testing
78    let partyid= args.party_id.clone();
79    let default_threads = rayon::current_num_threads() / 2;
80
81
82    tracing::info!("setting up network");
83    let mut nets: Vec<TcpNetwork> = Vec::new();
84    let mut states: Vec<Rep3State> = Vec::new();
85
86    for i in 0..args.threads {
87        let file_path = PathBuf::from(format!("{}{}/config_party{}.toml", args.config_dir.display(), i, partyid));
88        let config: NetworkConfig =toml::from_str(&std::fs::read_to_string(file_path).context("opening config file")?).context("parsing config file")?;
89        let net = TcpNetwork::new(config)?;
90        let state = Rep3State::new(&net)?;
91        nets.push(net);
92        states.push(state);
93    }
94
95    if states.len() <= default_threads {
96        let diff = default_threads - states.len();
97        for _i in 0..diff{
98            let state = states[0].fork(0)?;
99            states.push(state);
100        }
101    }
102    
103    let nets = nets.iter().collect::<Vec<&TcpNetwork>>();
104    let mut states = states.iter_mut().collect::<Vec<&mut Rep3State>>();
105    let party_id = states[0].id;
106
107    let mut mpc_exec_args = NetStateArgs::new(
108        &nets,
109        &mut states,
110    );
111
112    tracing::info!("Network setup completed");
113    
114    
115    let (lineitem_table, lineitem_table_polars) = tpch_database_gen::gen_lineitem_table(sf, &mut mpc_exec_args)?;
116    tracing::info!("Lineitem table generated with {} rows", lineitem_table.num_rows());
117
118    let (customer_table, customer_table_polars) = tpch_database_gen::gen_customer_table(sf, &mut mpc_exec_args)?;
119    tracing::info!("Customer table generated with {} rows", customer_table.num_rows());
120
121    let (mut orders_table, orders_table_polars) = tpch_database_gen::gen_orders_table(sf, &mut mpc_exec_args)?;
122    tracing::info!("Orders table generated with {} rows", orders_table.num_rows());
123
124    let (supplier_table, supplier_table_polars) = tpch_database_gen::gen_supplier_table(sf, &mut mpc_exec_args)?;
125    tracing::info!("Supplier table generated with {} rows", supplier_table.num_rows());
126
127    let (mut region_table, region_table_polars) = tpch_database_gen::gen_region_table(&mut mpc_exec_args)?;
128    tracing::info!("Region table generated with {} rows", region_table.num_rows());
129
130    let (nation_table, nation_table_polars) = tpch_database_gen::gen_nation_table(&mut mpc_exec_args)?;
131    tracing::info!("Nation table generated with {} rows", nation_table.num_rows());
132
133    tracing::info!("converting some columns to binary");
134
135    let o_orderdate_binary = orders_table["o_orderdate"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
136    orders_table.insert_column("[o_orderdate]".to_string(), o_orderdate_binary);
137
138    let r_name_binary = region_table["r_name"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
139    region_table.insert_column("[r_name]".to_string(), r_name_binary);
140
141    
142    tracing::info!("Projecting tables");
143    /* 
144    Customer.project({"[CustKey]", "[NationKey]"});
145    Orders.project({"[OrderKey]", "[CustKey]", "[OrderDate]"});
146    Lineitem.project({"[OrderKey]", "[SuppKey]", "ExtendedPrice", "Discount"});
147    Supplier.project({"[SuppKey]", "[NationKey]"});
148    Nation.project({"[NationKey]", "[RegionKey]", "[Name]"});
149    Region.project({"[RegionKey]", "[Name]"});
150    */
151
152    let l_col_names = vec!["l_orderkey", "l_suppkey", "l_extendedprice", "l_discount", "valid"];
153    let lineitem_table = lineitem_table.project(l_col_names)?;
154
155    let o_col_names = vec!["o_orderkey", "o_custkey", "[o_orderdate]", "valid"];
156    let mut orders_table = orders_table.project(o_col_names)?;
157
158    let c_col_names = vec!["c_custkey", "c_nationkey", "valid"];
159    let customer_table = customer_table.project(c_col_names)?;
160
161    let s_col_names = vec!["s_suppkey", "s_nationkey", "valid"];
162    let supplier_table = supplier_table.project(s_col_names)?;
163
164    let n_col_names = vec!["n_nationkey", "n_regionkey", "n_name", "valid"];
165    let nation_table = nation_table.project(n_col_names)?;
166
167    let r_col_names = vec!["r_regionkey", "[r_name]", "valid"];
168    let mut region_table = region_table.project(r_col_names)?;
169    
170    tracing::info!("Projection completed");
171
172
173    tracing::info!("Q5 start");
174    let tot_start = Instant::now();
175
176    tracing::info!("o_orderdate >= date '[DATE]' and o_orderdate < date '[DATE]' + interval '1' year");
177
178    let c1 = orders_table["[o_orderdate]"].ge_public_binary(&DATE, &mut mpc_exec_args)?;
179    let c2 = orders_table["[o_orderdate]"].lt_public_binary(&(DATE + DATE_INTERVAL), &mut mpc_exec_args)?;
180    let c = c1.and(&c2, &mut mpc_exec_args)?;
181
182    let _ = orders_table.filter_directed_by_bool(c.get_data(), &mut mpc_exec_args)?;
183    //let _ = orders_table.filter_public("[o_orderdate]", Predicate::GreaterOrEqualBinary, &DATE, &nets, &mut states)?;
184    //let _ = orders_table.filter_public("[o_orderdate]", Predicate::LessThanBinary, &(DATE + DATE_INTERVAL), &nets, &mut states)?;
185
186    orders_table.delete_column("[o_orderdate]");
187
188
189
190    tracing::info!("r_name = '[REGION]'");
191
192    let _ = region_table.filter_public("[r_name]", Predicate::EqualBinary, &REGION, &mut mpc_exec_args)?;
193
194    region_table.delete_column("[r_name]");
195
196
197    tracing::info!("sub query 1: s_nationkey = n_nationkey and n_regionkey = r_regionkey");
198
199    let region_nation_table = region_table.inner_join(
200        "r_regionkey",
201        "n_regionkey",
202        &nation_table,
203        &mut mpc_exec_args,
204    )?;
205
206    let region_nation_supplier_table = region_nation_table.inner_join(
207        "n_nationkey",
208        "s_nationkey",
209        &supplier_table,
210        &mut mpc_exec_args,
211    )?;
212
213    
214    let sub1_table = region_nation_supplier_table.inner_join(
215        "s_suppkey",
216        "l_suppkey",
217        &lineitem_table,
218        &mut mpc_exec_args,
219    )?;
220
221
222    tracing::info!("sub query 2: c_custkey = o_custkey");
223
224    let sub2_table = customer_table.inner_join(
225        "c_custkey",
226        "o_custkey",
227        &orders_table,
228        &mut mpc_exec_args,
229    )?;
230
231
232    tracing::info!("merge tree: l_orderkey = o_orderkey");
233
234    let mut final_table = sub2_table.inner_join(
235        "o_orderkey",
236        "l_orderkey",
237        &sub1_table,
238        &mut mpc_exec_args,
239    )?;
240
241    
242    //tracing::info!("c_nationkey = s_nationkey where this is a filter condition");
243
244    let _ = final_table.filter_shared("c_nationkey", "s_nationkey", Predicate::Equal, &mut mpc_exec_args);
245
246    
247    tracing::info!("Computing revenue");
248
249    let const_element = RingElement(100u64);
250    
251    let mut revenue: ShareColumn<Rep3RingShare<u64>> = final_table["l_extendedprice"].clone() * 
252                                                (&(-final_table["l_discount"].clone() + (const_element, &party_id)), &mut mpc_exec_args) 
253                                                / (&const_element, &mut mpc_exec_args);
254
255    revenue.update_name("l_revenue".to_string());
256    final_table.insert_column("l_revenue".to_string(), revenue);
257
258    final_table.delete_column("l_discount");
259    final_table.delete_column("l_extendedprice");
260
261     
262    tracing::info!("group by n_name");
263
264    let group_by_col_names = vec!["n_name"];
265
266    let (e,perm,_) = final_table.group_by(
267        group_by_col_names,
268        &mut mpc_exec_args,
269    )?;
270
271
272    tracing::info!("table aggregate");
273
274    let _ = final_table.agg_sum(
275        "l_revenue",
276        "revenue",
277        &e,
278        &perm,
279        &mut mpc_exec_args,
280    )?;
281
282    /* 
283    tracing::info!("secure cut lineorder_table size to nation_table size");
284    final_table.head(get_nation_table_size() as usize);
285    */
286
287    tracing::info!("order by revenue desc");
288
289    let _ = final_table.order_by(
290        "revenue",
291        false,
292        &mut mpc_exec_args,
293    )?;
294    
295
296    tracing::info!("Q5 no-secure-cut execution completed");
297
298    if party_id == PartyID::ID0 {
299        tracing::info!("Total Q5 no-secure-cut execution time: {:?}", tot_start.elapsed());
300    }
301    print_communication_stats(&mpc_exec_args, "Q5 no-secure-cut");
302
303
304
305//************* polars verification *************//
306
307    let _ = final_table.order_by("valid", false, &mut mpc_exec_args)?;
308    
309    let mut result_table = final_table.project(vec!["n_name", "revenue", "valid"])?;
310
311    let sum_valid = result_table["valid"].prefix_sum();
312    let open_valid = open(sum_valid, mpc_exec_args.nets[0])?.0;
313    tracing::info!("open valid: {}", open_valid);
314    result_table.head(open_valid as usize);
315
316    let mpc_result = result_table.open(&mut mpc_exec_args)?;
317
318    if party_id == PartyID::ID0 {
319        tracing::info!("Q5 polars:");
320
321        let region = region_table_polars.unwrap();
322        let nation = nation_table_polars.unwrap();
323        let supplier = supplier_table_polars.unwrap();
324        let lineitem = lineitem_table_polars.unwrap();
325        let orders = orders_table_polars.unwrap();
326        let customer = customer_table_polars.unwrap();
327
328        let q5_sub1 = region.clone().lazy()
329        .filter(col("r_name").eq(lit(REGION)))
330        .join(
331            nation.clone().lazy(),
332            [col("r_regionkey")],
333            [col("n_regionkey")],
334            JoinArgs::new(JoinType::Inner)
335        )
336        .join(
337            supplier.clone().lazy(),
338            [col("n_nationkey")],
339            [col("s_nationkey")],
340            JoinArgs::new(JoinType::Inner)
341        )
342        .join(
343            lineitem.clone().lazy(),
344            [col("s_suppkey")],
345            [col("l_suppkey")],
346            JoinArgs::new(JoinType::Inner)
347        );
348
349        
350        let q5_final = customer.clone().lazy()
351        .join(
352            orders.clone().lazy(),
353            [col("c_custkey")],
354            [col("o_custkey")],
355            JoinArgs::new(JoinType::Inner)
356        ).join(
357            q5_sub1.lazy(),
358            [col("o_orderkey")],
359            [col("l_orderkey")],
360            JoinArgs::new(JoinType::Inner)
361        ).filter(
362            col("c_nationkey").eq(col("n_nationkey"))
363        ).filter(
364            col("o_orderdate").gt_eq(lit(DATE)).and(col("o_orderdate").lt(lit(DATE + DATE_INTERVAL)))
365        ).with_column(
366            (((col("l_extendedprice") * (lit(100) - col("l_discount"))) / lit(100))).alias("revenue_contribution")
367        )
368        .group_by([col("n_name")])
369        .agg([
370            col("revenue_contribution").sum().alias("revenue")
371        ])
372        .sort(["revenue"], SortMultipleOptions::default().with_order_descending(true))
373        .collect()
374        .unwrap();
375
376
377        let _q5_result = region.lazy()
378        .filter(col("r_name").eq(lit(REGION)))
379        .join(
380            nation.lazy(),
381            [col("r_regionkey")],
382            [col("n_regionkey")],
383            JoinArgs::new(JoinType::Inner)
384        )
385        .join(
386            supplier.lazy(),
387            [col("n_nationkey")],
388            [col("s_nationkey")],
389            JoinArgs::new(JoinType::Inner)
390        )
391        .join(
392            lineitem.lazy(),
393            [col("s_suppkey")],
394            [col("l_suppkey")],
395            JoinArgs::new(JoinType::Inner)
396        )
397        .join(
398            orders.lazy(),
399            [col("l_orderkey")],
400            [col("o_orderkey")],
401            JoinArgs::new(JoinType::Inner)
402        )
403        .filter(
404            col("o_orderdate").gt_eq(lit(DATE)).and(col("o_orderdate").lt(lit(DATE + DATE_INTERVAL)))
405        )
406        .join(
407            customer.lazy(),
408            [col("o_custkey"), col("n_nationkey")], // s_nationkey might be dropped, use n_nationkey
409            [col("c_custkey"), col("c_nationkey")],
410            JoinArgs::new(JoinType::Inner)
411        )
412        .with_column(
413            (col("l_extendedprice") * ((lit(100) - col("l_discount")) / lit(100))).alias("revenue_contribution")
414        )
415        .group_by([col("n_name")])
416        .agg([
417            col("revenue_contribution").sum().alias("revenue")
418        ])
419        .sort(["revenue"], SortMultipleOptions::default().with_order_descending(true))
420        .collect()
421        .unwrap();
422  
423        let mpc_n_name = mpc_result["n_name"].get_data();
424        let mpc_revenue = mpc_result["revenue"].get_data();
425
426        //let polars_n_name_1 = q5_result.column("n_name")?.u64()?.into_no_null_iter().collect::<Vec<_>>();
427        let polars_n_name = q5_final.column("n_name")?.u64()?.into_no_null_iter().collect::<Vec<_>>();
428        let polars_revenue = q5_final.column("revenue")?.u64()?.into_no_null_iter().collect::<Vec<_>>();
429        
430
431        //println!("MPC n_name: {:?}", mpc_n_name);
432        //println!("Polars n_name: {:?}", polars_n_name);
433        //println!("MPC revenue: {:?}", mpc_revenue);
434        //println!("Polars revenue: {:?}", polars_revenue);
435        //println!("MPC valid: {:?}", mpc_valid);
436        //assert_eq!(&polars_n_name_1, &polars_n_name);
437        tracing::info!("rows of polars_n_name: {:?}", polars_n_name.len());
438        assert_eq!(mpc_n_name, &polars_n_name);
439        assert_eq!(mpc_revenue, &polars_revenue);
440
441        tracing::info!("Q5 Passed: Qamboo result MATCHES Polars result !");
442    }
443    
444    Ok(())
