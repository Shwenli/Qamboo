2// Rcdiff Query from the SMCQL and Secrecy paper
3/*
4 * SQL:
5 * WITH rcd AS (
6 *      SELECT pid, time, row_no
7 *      FROM diagnosis
8 *      WHERE diag=cdiff)
9 *    SELECT DISTINCT pid
10 *    FROM rcd r1 JOIN rcd r2
11 *    WHERE r1.pid = r2.pid
12 *       AND r2.time - r1.time >= 15 DAYS
13 *       AND r2.time - r1.time <= 56 DAYS
14 *       AND r2.row_no = r1.row_no + 1
15 *
16 * Note: row_no is not present in our schema; it is an implicit row generated
17 * after sorting.
18 *
19 */
20

46const CDIFF_DIAG: u64 = 8;
47const TIME_MIN: u64 = 15;
48const TIME_MAX: u64 = 56;
49
50
51
52#[derive(Parser)]
53struct Args {
54    /// The config file path
55    #[clap(short = 'c', long, value_name = "CONFIGDIR")]
56    config_dir: PathBuf,
57    
58    /// The party ID (0, 1, 2)
59    #[clap(short = 'p', long, value_name = "PARTYID", default_value = "0")]
60    party_id: String,
61
62    /// The number of threads
63    #[clap(short = 't', long, value_name = "THREADS", default_value = "6")]
64    threads: usize,
65
66    /// Scale factor
67    #[clap(short = 's', long, value_name = "SF", default_value = "0.01")]
68    sf: f32,
69}
70
71
72fn main() -> Result<()> {
73    let args = Args::parse();
74    install_tracing();
75
76    let sf = args.sf;
77    let partyid = args.party_id.clone();
78    let default_threads = rayon::current_num_threads() / 2;
79
80    tracing::info!("setting up network");
81    let mut nets: Vec<TcpNetwork> = Vec::new();
82    let mut states: Vec<Rep3State> = Vec::new();
83
84    for i in 0..args.threads {
85        let file_path = PathBuf::from(format!("{}{}/config_party{}.toml", args.config_dir.display(), i, partyid));
86        let config: NetworkConfig = toml::from_str(&std::fs::read_to_string(file_path).context("opening config file")?).context("parsing config file")?;
87        let net = TcpNetwork::new(config)?;
88        let state = Rep3State::new(&net)?;
89        nets.push(net);
90        states.push(state);
91    }
92
93    if states.len() <= default_threads {
94        let diff = default_threads - states.len();
95        for _i in 0..diff {
96            let state = states[0].fork(0)?;
97            states.push(state);
98        }
99    }
100
101    let nets = nets.iter().collect::<Vec<&TcpNetwork>>();
102    let mut states = states.iter_mut().collect::<Vec<&mut Rep3State>>();
103    let party_id = states[0].id;
104
105    let mut mpc_exec_args = NetStateArgs::new(&nets, &mut states);
106    
107    tracing::info!("Network setup completed");
108
109
110    tracing::info!("Generating tables");
111    let (mut rcd_table, rcd_polars) = gen_diagnosis_table(sf, &mut mpc_exec_args)?;
112    tracing::info!("Diagnosis table generated with {} rows", rcd_table.num_rows());
113
114    tracing::info!("converting some columns to binary");
115    let rcd_diag_binary = rcd_table["diag"].add_new_col_from_arithmetic_to_binary(&mut mpc_exec_args)?;
116    rcd_table.insert_column("[diag]".to_string(), rcd_diag_binary);
117
118
119    tracing::info!("Rcdiff query start");
120    let tot_start = Instant::now();
121
122    tracing::info!("Filtering rcd table for CDIFF diagnosis");
123    let _ = rcd_table.filter_public("[diag]", Predicate::EqualBinary, &CDIFF_DIAG, &mut mpc_exec_args);
124    rcd_table.delete_column("[diag]");
125
126
127    tracing::info!("Sorting rcd table by pid and time");
128    let _ = rcd_table.order_by("time", true, &mut mpc_exec_args);
129    let _ = rcd_table.order_by("pid", true, &mut mpc_exec_args);
130    let _ = rcd_table.order_by("valid", false, &mut mpc_exec_args);
131
132
133    tracing::info!("WHERE r1.pid = r2.pid AND r2.time - r1.time >= 15 DAYS AND r2.time - r1.time <= 56 DAYS AND r2.row_no = r1.row_no + 1");
134    let mut pid2_data = rcd_table["pid"].get_data()[1..rcd_table.num_rows()].to_vec();
135    pid2_data.push(get_data_share(&(1u64<<63u64), party_id)?);
136    rcd_table.insert_column("pid2".to_string(), ShareColumn::new(pid2_data,ShareType::Arithmetic, "pid2".to_string()));
137    
138    let mut time2 = rcd_table["time"].get_data()[1..rcd_table.num_rows()].to_vec();
139    time2.push(get_data_share(&(1u64<<63u64), party_id)?);
140    rcd_table.insert_column("time2".to_string(), ShareColumn::new(time2, ShareType::Arithmetic, "time2".to_string()));
141
142    let time_diff = rcd_table["time2"].clone() - rcd_table["time"].clone();
143    rcd_table.insert_column("time_diff".to_string(), time_diff);
144
145    let c1 = rcd_table["time_diff"].ge_public(&TIME_MIN, &mut mpc_exec_args)?;
146    let c2 = rcd_table["time_diff"].le_public(&TIME_MAX, &mut mpc_exec_args)?;
147    let c3 = rcd_table["pid"].equal(&rcd_table["pid2"], &mut mpc_exec_args)?;
148    let condition = c1.and(&c2, &mut mpc_exec_args)?.and(&c3, &mut mpc_exec_args)?;
149    let _ = rcd_table.filter_directed_by_bool(condition.get_data(), &mut mpc_exec_args)?;
150
151
152    tracing::info!("Selecting distinct pid");
153    let _ = rcd_table.group_by(vec!["pid"], &mut mpc_exec_args)?;
154    let mut final_table = rcd_table.project(vec!["pid", "valid"])?;
155
156
157    tracing::info!("rcdiff query execution completed");
158    if party_id == PartyID::ID0 {
159        tracing::info!("Total execution time: {:?}", tot_start.elapsed());
160    }
161    print_communication_stats(&mpc_exec_args, "rcdiff");
162
163
164
165
166//************* polars verification *************//
167
168    let sum_valid = final_table["valid"].prefix_sum();
169    let open_sum_valid = open(sum_valid, mpc_exec_args.nets[0])?;
170    final_table.head(open_sum_valid.0 as usize);
171
172    let mpc_result = final_table.open(&mut mpc_exec_args)?;
173
174    if party_id == PartyID::ID0 {
175        tracing::info!("Rcdiff polars verification:");
176
177        let diagnosis = rcd_polars.unwrap();
178        
179        let rcd_result = diagnosis.lazy()
180            .filter(col("diag").eq(lit(CDIFF_DIAG)))
181            .sort(
182                ["pid", "time"], 
183                SortMultipleOptions::default().with_order_descending_multi([false, false])
184            )
185            .with_column(
186                col("time").shift(lit(-1)).over([col("pid")]).alias("next_time")
187            )
188            .filter(col("next_time").is_not_null())
189            .with_column(
190                (col("next_time") - col("time")).alias("time_diff")
191            )
192            .filter(
193                col("time_diff").gt_eq(lit(TIME_MIN))
194                .and(col("time_diff").lt_eq(lit(TIME_MAX)))
195            )
196            .select([col("pid")])
197            .unique(None, UniqueKeepStrategy::First) // 模拟最后的 group_by(vec!["pid"])
198            .sort(["pid"], SortMultipleOptions::default()) // 升序保证断言通过
199            .collect()?;
200
201        tracing::info!("Polars distinct pid count: {}", rcd_result.height());
202        
203        let polars_pids = rcd_result.column("pid")?.u64()?.into_no_null_iter().collect::<Vec<u64>>();
204        let mpc_pids = mpc_result["pid"].get_data().to_vec();
205
206        assert_eq!(mpc_pids, polars_pids, "Distinct pid count does not match!");
207
208        tracing::info!("rcdiff verification passed!");
209    }
210
211
212    Ok(())
213}
